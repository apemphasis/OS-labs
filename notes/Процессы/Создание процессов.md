```C++
BOOL CreateProcess (
	LPCTSTR lpApplicationName, 
	LPTSTR  lpCommandLine, 
	LPSECUROTY_ATTRIBUTES lpProcessAttributes, 
	LPSECURITY_ATTRIBUTES lpThreadAttribures, 
	BOOL    bInheritHandles, 
	DWORD   dwCreationFlags, 
	LPVOID  lpEnvironment, 
	LPCTSTR lpCurrentDirectory, 
	LPSTARTUPINFO  lpStartUpInfo, 
	LPPROCESS_INFORMATION lpProcessInformation 
);
```
Пример 1:
```C++
char lpszAppName[] = "C:\\process.exe";  // full path
STARTUPINFO si;
PROCESS_INFORMATION piApp;

ZeroMemory(&si, sizeof(STARTUPINFO));   // занулить si
si.cb = sizeof(STARTUPINFO);

if (!CreateProcess(lpszAppName, NULL, NULL, NULL, FALSE, CREATE_NEW_CONSOLE, NULL, NULL, &si, &piApp)){
	return GetLastError();
}

CloseHandle(piApp.hTread);
CloseHandle(piApp.hProcess);
```
Пример 2:
```C++
char lpszCommandLine[] = "process.exe p1 p2 p3";

STARTUPINFO si;
PROCESS_INFORMATION piCom;

ZeroMemory(&si, sizeof(STARTUPINFO));
si.cb = sizeof(STARTUPINFO);

CreateProcess(NULL, lpszCommandLine, NULL, NULL, FALSE, CREATE_NEW_CONSOLE, NULL, NULL, &si, &piCom);

CloseHandle(piCom.hThread);
CloseHandle(piCom.hProcess);


```

```C++
typedef struct _PROCESS_INFORMATION { 
	HANDLE hProcess; 
	HANDLE hThread; 
	DWORD dwProcessId; 
	DWORD dwThreadId; 
} PROCESS_INFORMATION, *PPROCESS_INFORMATION, *LPPROCESS_INFORMATION;
```

```C++
typedef struct _STARTUPINFO {                     // si
DWORD cb;                  // размер структуры в байтах
LPTSTR lpReserved;         
LPTSTR lpDesktop;          
LPTSTR lpTitle;
DWORD dwX;
DWORD dwY;
DWORD dwXSize;
DWORD dwYSize;
DWORD dwXCountChars;
DWORD dwYCountChars;
DWORD dwFillAttribute;    // FOREGROUND_[COLOR] | BACKGROUND_[COLOR]
DWORD dwFlags;
WORD wShowWindow;
WORD cbReserved2;
LPBYTE lpReserved2;
HANDLE hStdInput;
HANDLE hStdOutput;
HANDLE hStdError;

} STARTUPINFO, *LPSTARTUPINFO;
```

| dwFlags value           | Description                                                            |
| ----------------------- | ---------------------------------------------------------------------- |
| STARTF_FORCEONFEEDBACK  | -                                                                      |
| STARTF_FORCEOFFFEEDBACK | -                                                                      |
| STARTF_RUNFULLSCREEN    | CLEAR                                                                  |
| STARTF_USECOUNTCHARS    | for dwXCountChars and dwYCountChars                                    |
| STARTF_USEFILLATTRIBUTE | for dwFillAttribute                                                    |
| STARTF_USEPOSITION      | for dwX and dwY                                                        |
| STARTF_USESHOWWINDOW    | for wShowWindow -                                                      |
| STARTF_USESIZE          | for dwXSize and dwYSize                                                |
| STARTF_USESTRHANDLES    | for hStdInput, hStdOutput and hStdError (fInheritHandles must be TRUE) |
